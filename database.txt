-- 1. Proje için şema (kullanıcı) oluşturalım
CREATE USER c##career_planning_assistant IDENTIFIED BY Oracle1234;

-- 2. Gerekli yetkileri verelim
GRANT CONNECT, RESOURCE, DBA TO c##career_planning_assistant;

-- 1. Kullanıcılar Tablosu (Sisteme kayıt olan öğrenciler)
CREATE TABLE c##career_planning_assistant.users (
    user_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    first_name VARCHAR2(50) NOT NULL,
    last_name VARCHAR2(50) NOT NULL,
    email VARCHAR2(100) UNIQUE NOT NULL,
    password_hash VARCHAR2(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Sınav Puanları ve Netler Tablosu (Modül 7 ve 10 için)
CREATE TABLE c##career_planning_assistant.exam_scores (
    score_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id NUMBER REFERENCES c##career_planning_assistant.users(user_id),
    tyt_net FLOAT,
    ayt_net FLOAT,
    obp_puani FLOAT,
    calculated_placement_score FLOAT,
    exam_year NUMBER(4),
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Bölümler ve Meslekler Havuzu (Modül 1, 3, 4, 9, 10 için ana tablo)
CREATE TABLE c##career_planning_assistant.departments (
    dept_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dept_name VARCHAR2(150) NOT NULL,
    dept_type VARCHAR2(20) CHECK (dept_type IN ('Lisans', 'Önlisans')),
    difficulty_level VARCHAR2(20), -- Zorluk derecesi
    work_environment VARCHAR2(100), -- Çalışma ortamı (Ofis, Saha vb.)
    salary_potential VARCHAR2(50), -- Maaş potansiyeli
    advantages CLOB, -- Avantajlar
    disadvantages CLOB, -- Dezavantajlar
    future_trend_score NUMBER(3,2) -- Modül 1: Gelecek trend skoru (0-10 arası)
);

-- 4. Anket ve Yetenek Testi Soruları (Modül 2 için)
CREATE TABLE c##career_planning_assistant.questions (
    question_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    question_text VARCHAR2(500) NOT NULL,
    category VARCHAR2(50) -- Sayısal, Sözel, Eşit Ağırlık, Yaratıcı vb.
);

-- 5. Kullanıcı Test Yanıtları ve Skorları (Modül 2 için)
CREATE TABLE c##career_planning_assistant.user_test_results (
    result_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id NUMBER REFERENCES c##career_planning_assistant.users(user_id),
    category_name VARCHAR2(50),
    score_percentage NUMBER(5,2)
);

-- 6. Rol Simülasyonu - "Günüm Nasıl Geçer?" (Modül 5 için)
CREATE TABLE c##career_planning_assistant.role_simulations (
    sim_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dept_id NUMBER REFERENCES c##career_planning_assistant.departments(dept_id),
    time_slot VARCHAR2(50), -- Örn: "09:00 - Sabah Toplantısı"
    task_description VARCHAR2(300)
);

-- 7. Eksik Beceri Tamamlama Yol Haritası (Modül 8 için)
CREATE TABLE c##career_planning_assistant.skill_gaps (
    skill_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dept_id NUMBER REFERENCES c##career_planning_assistant.departments(dept_id),
    required_skill VARCHAR2(150),
    recommended_resource VARCHAR2(255) -- Kurs, kitap veya pratik önerisi
);
